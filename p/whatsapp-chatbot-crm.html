Source Code: Webhook Auto-Logger

Berikut script Node.js minimal untuk menangkap pesan WhatsApp dan melemparnya ke Google Sheets CRM secara real-time:

server.js (WhatsApp Webhook Handler) 
const express = require('express');
const axios = require('axios');
const app = express();
app.use(express.json());

// Endpoint untuk menerima webhook dari WhatsApp Client
app.post('/webhook/whatsapp', async (req, res) => {
  const { sender, message, pushName, timestamp } = req.body;
  
  console.log(`[INCOMING] ${pushName} (${sender}): ${message}`);

  // 1. Klasifikasi Intent sederhana berdasarkan keyword
  let tag = 'GENERAL_INQUIRY';
  if (/harga|biaya|pricelist|paket/i.test(message)) tag = 'HOT_PRICING';
  if (/audit|jadwal|konsul|call/i.test(message)) tag = 'HIGH_INTENT_AUDIT';
  if (/wacrm|template|gratis/i.test(message)) tag = 'LEAD_MAGNET_CLAIM';

  // 2. Kirim ke Google Apps Script / Google Sheets CRM
  const GOOGLE_SHEETS_WEBHOOK = "https://script.google.com/macros/s/YOUR_SCRIPT_ID/exec";
  
  try {
    await axios.post(GOOGLE_SHEETS_WEBHOOK, {
      name: pushName,
      phone: sender,
      message: message,
      intentTag: tag,
      receivedAt: new Date(timestamp || Date.now()).toISOString()
    });
    
    res.status(200).json({ status: 'success', tagLogged: tag });
  } catch (err) {
    console.error('Error logging to Sheets:', err.message);
    res.status(500).json({ error: err.message });
  }
});

app.listen(3000, () => console.log('WA CRM Webhook running on port 3000'));
